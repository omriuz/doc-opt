import argparse
import pickle


parser = argparse.ArgumentParser()
parser.add_argument("--test_case", type=int, default="1")
args = parser.parse_args()
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split

features_dataframe = pickle.load(open(f"input/input{args.test_case}.pkl", "rb"))
### BEGIN SOLUTION
n = features_dataframe.shape[0]
train_size = 0.8
test_size = 1 - train_size + 0.005
train_dataframe = features_dataframe.iloc[int(n * test_size):]
test_dataframe = features_dataframe.iloc[:int(n * test_size)]
### END SOLUTION
with open('result/result_{}.pkl'.format(args.test_case), 'wb') as f:
    pickle.dump((train_dataframe, test_dataframe), f)
