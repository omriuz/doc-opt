import pickle
import argparse

parser = argparse.ArgumentParser()
parser.add_argument("--test_case", type=int, default=1)
args = parser.parse_args()
import scipy.optimize
import numpy as np
a, x_true, y, x0, x_lower_bounds = pickle.load(open(f"input/input{args.test_case}.pkl", "rb"))
def residual_ans(x, a, y):
    s = ((y - a.dot(x**2))**2).sum()
    return s
bounds = [[x, None] for x in x_lower_bounds]
out = scipy.optimize.minimize(residual_ans, x0=x0, args=(a, y), method= 'L-BFGS-B', bounds=bounds).x
#print(out)


with open('result/result_{}.pkl'.format(args.test_case), 'wb') as f:
    pickle.dump(out, f)
